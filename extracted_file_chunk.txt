21: function getDistance(lat1: number, lon1: number, lat2: number, lon2: number) {
22:   const R = 6371;
23:   const dLat = (lat2 - lat1) * Math.PI / 180;
24:   const dLon = (lon2 - lon1) * Math.PI / 180;
25:   const a =
26:     Math.sin(dLat / 2) * Math.sin(dLat / 2) +
27:     Math.cos(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) *
28:     Math.sin(dLon / 2) * Math.sin(dLon / 2);
29:   const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
30:   return R * c * 1000;
31: }
